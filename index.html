import express from 'express'; import path from 'path'; import { fileURLToPath } from 'url'; const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(__dirname)); app.post('/api/contact', async (req, res) => { console.log('Received contact submission:', req.body); // כתובת ה-n8n Webhook const DEFAULT_N8N_URL = 'https://pasternak.app.n8n.cloud/webhook/contact'; const webhookUrl = process.env.N8N_WEBHOOK_URL || process.env.WEBHOOK_URL || DEFAULT_N8N_URL; const { name, email, message, _gotcha } = req.body; // מנגנון הגנה מפני ספאם (Honeypot) if (_gotcha) { console.log('Spam submission detected via honeypot field.'); return res.json({ success: true, message: 'Message received' }); } if (!name || !email || !message) { return res.status(400).json({ success: false, error: 'All fields are required' }); } let webhookSuccess = false; let webhookError = null; if (webhookUrl && webhookUrl.trim() !== '') { try { console.log(`Forwarding contact form submission to n8n webhook: ${webhookUrl}`); const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 8000); // 8 שניות Timeout const response = await fetch(webhookUrl.trim(), { method: 'POST', headers: { 'Content-Type': 'application/json', 'User-Agent': 'Hadas-Pasternak-Website/1.0' }, body: JSON.stringify({ name: name.trim(), email: email.trim(), message: message.trim(), submittedAt: new Date().toISOString(), source: 'Hadas Pasternak Website' }), signal: controller.signal }); clearTimeout(timeoutId); if (response.ok) { webhookSuccess = true; console.log(`Successfully delivered to n8n (HTTP ${response.status})`); } else { const errText = await response.text(); webhookError = `n8n responded with HTTP ${response.status}: ${errText}`; console.error(webhookError); } } catch (err) { webhookError = err.name === 'AbortError' ? 'n8n webhook request timed out (8s)' : err.message; console.error('Error delivering to n8n webhook:', webhookError); } } res.json({ success: true, message: 'Form submission processed successfully', webhookConfigured: true, webhookDelivered: webhookSuccess, webhookError: webhookError }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'index.html')); }); app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
